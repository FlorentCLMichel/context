\startuseMPgraphic{page}
    w := OverlayWidth /10 ;
    h := OverlayHeight/10 ;
    set_grid(10,10,1,1) ;
    forever :
        if new_on_grid(uniformdeviate 10, uniformdeviate 10) :
            fill (fullsquare xyscaled (w,h) shifted (w*round(dx),h*round(dy)))
                enlarged 5pt randomized 10pt
                withcolor transparent (1,.5,white randomized (.5,.8)) ;
        fi ;
        exitif grid_full ;
    endfor ;
\stopuseMPgraphic

\startuseMPgraphic{text}
    fill (fullsquare xyscaled (OverlayWidth,OverlayHeight))
        enlarged 5pt randomized 10pt
        withcolor transparent (1,.5, red randomized (.3,.5)) ;
\stopuseMPgraphic

\startuseMPgraphic{back}
    fill (fullsquare xyscaled (OverlayWidth,OverlayHeight))
        enlarged 10pt randomized 10pt
        withcolor \MPcolor{shade-8} ;
\stopuseMPgraphic

\usemodule[mathml]

\startxmlsetups xml:graphics
    \xmlfilter{#1}{/introduction/command(xml:introduction)}
    \xmlfilter{#1}{/graphic/command(xml:graphic)}
    \setupbackgrounds[page][background=]
    \xmlfilter{#1}{/graphic/command(xml:biggraphic)}
\stopxmlsetups

\startxmlsetups xml:introduction
    \startstandardmakeup
    \vfill
    \startframedtext[none]
        \ignorespaces
        \xmlfirst{#1}{/description/summary}
        \removeunwantedspaces
    \stopframedtext
    \stopstandardmakeup
\stopxmlsetups

\startxmlsetups xml:graphic
    \startstandardmakeup
    \vfill
    \writestatus{graphic}{1 : \xmltext{#1}{/resources/snippet}}
    \setlayer
        [graphic]
        [x=40pt,y=40pt,location=rb]
        {
            \pagereference[\xmltext{#1}{/resources/result}:text]
            \gotobox {
                \externalfigure
                    [\xmltext{#1}{/resources/snippet}.pdf]
                    [background=back,
                     factor=auto,
                     compact=no,
                     maxwidth=.7\paperwidth,
                     maxheight=.7\paperheight]
            }
            [\xmltext{#1}{/resources/result}:page]
        }
    \startframedtext[none]
        \ignorespaces
        \xmlfirst{#1}{/description/summary}
        \removeunwantedspaces
    \stopframedtext
    \stopstandardmakeup
\stopxmlsetups

\startxmlsetups xml:biggraphic
    \startTEXpage[align=]% hm, needed, as we get wrong dimensions otherwise ... bug?
        \writestatus{graphic}{2 : \xmltext{#1}{/resources/snippet}}
        \dontcomplain
        \dontleavehmode\pagereference[\xmltext{#1}{/resources/result}:page]%
        \gotobox {
            \externalfigure
              [\xmltext{#1}{/resources/result}.pdf]
              [compact=no]
          }
          [\xmltext{#1}{/resources/result}:text]%
    \stopTEXpage
\stopxmlsetups

\startxmlsetups xml:mysetups

\stopxmlsetups
