% a bit of XML

\startxmlsetups xml:logo
    \getvalue{\cldcontext{string.upper("\xmlatt{#1}{label}")}}
\stopxmlsetups

\startxmlsetups xml:v
    \dontleavehmode
    \begingroup
        \tt
        \xmlflush{#1}
    \endgroup
\stopxmlsetups

\startxmlsetups xml:p
    \startparagraph
        \ignorespaces
            \xmlflush{#1}
        \removeunwantedspaces
    \stopparagraph
\stopxmlsetups

\startxmlsetups xml:compound
    \compound{-}
\stopxmlsetups

\startxmlsetups xml:destination
    \gotobox{\xmlflush{#1}}[file(\xmlattdef{#1}{destination}{unknown})]
\stopxmlsetups

\startxmlsetups xml:quote
    \quote{\xmlflush{#1}}
\stopxmlsetups

\startxmlsetups xml:basics
    \xmlsetsetup{main}{logo|v|p|compound|link|quote}{xml:*}
\stopxmlsetups

\xmlregistersetup{xml:basics}
