\startluacode

    local price = .50

    local function report(t)
        local usage = t.usage
        local total = usage == nil and t.days * t.runs * t.cycles * (t.seconds/3600) * (t.watts/1000) + 365 * 24 * (t.idling/1000)
        context.NC() context(t.days)
        context.NC() context(t.pages)
        context.NC() context(t.runs)
        context.NC() context(t.cycles)
        context.NC() context(t.seconds)
        context.NC()
        if t.pps then
            context(t.pps)
        else
            context("%0.1f",t.pages/t.seconds)
        end
        context.NC() context(t.watt)
        context.NC() context(t.idling)
        context.NC()
        if usage then
            context(usage)
            context.NC()
            context(t.price)
        else
            context("%0.3f",total)
            context.NC()
            context("%0.2f",total*price)
        end
        context.NC() context(t.comment)
        context.NC() context.NR()
    end

    context.starttabulate { "|r|r|r|r|r|r|r|r|r|r|l|" }
        context.HL()
        report {
            days = "days", runs = "runs", cycles = "cycles", seconds = "seconds", watts = "watts",
            usage = "kwh", idling = "idling", price = "euros", pages = "pages", pps ="pps", comment = "comment"
        }
        context.HL()
            report { days = 300, pages = 350, runs =    10, cycles = 2, seconds =  10, watts = 50, idling =   0, comment = "luametatex manual" }
            report { days = 300, pages = 200, runs =    50, cycles = 3, seconds =  15, watts = 50, idling =   0, comment = "m4all maintenance" }
            report { days = 100, pages = 400, runs =    20, cycles = 4, seconds =  30, watts = 50, idling = 200, comment = "m4all server" }
            report { days = 100, pages = 400, runs =    20, cycles = 4, seconds =  30, watts = 50, idling =  20, comment = "idem laptop" }
            report { days = 365, pages =  80, runs = 15000, cycles = 2, seconds =   5, watts = 50, idling = 200, comment = "nightly reports" }
            report { days = 365, pages =  80, runs = 15000, cycles = 2, seconds =   5, watts = 50, idling =  20, comment = "idem laptop" }
        context.HL()
    context.stoptabulate()

    context.blank()

    context("price per kwh: %0.2f",price)
