\def\MinW{1}
\def\MinH{1}
\def\MaxW{4}
\def\MaxH{4}

\def\Shift#1#2#3#4%
  {\vbox\bgroup
   \forgetall
   \offinterlineskip
   \def\CurrentN{#1}%
   \setupclipping[nx=4,ny=4]%
   \dorecurse{\MaxW}
     {\let\ClipW\recurselevel
      \dorecurse{\MaxH}
        {\let\ClipH\recurselevel
         \letvalue{clips-\ClipW-\ClipH-x}\ClipW
         \letvalue{clips-\ClipW-\ClipH-y}\ClipH}}%
   \def\CurrentW{#3}\let\PrevW\CurrentW
   \def\CurrentH{#4}\let\PrevH\CurrentH
   \ifnum#1>0
     \newcounter\CurrentI
     \doloop
       {\getrandomnumber\Random{0}{3}%
        \ifcase\Random\relax
          \ifnum\CurrentW<\MaxW\relax\increment\CurrentW\relax\fi \or
          \ifnum\CurrentW>\MinW\relax\decrement\CurrentW\relax\fi \or
          \ifnum\CurrentH<\MaxH\relax\increment\CurrentH\relax\fi \or
          \ifnum\CurrentH>\MinH\relax\decrement\CurrentH\relax\fi \or
        \fi
        \donefalse
        \ifnum\CurrentW=\PrevW\relax\else\donetrue\fi
        \ifnum\CurrentH=\PrevH\relax\else\donetrue\fi
        \ifdone
          \edef\TempW{\getvalue{clips-\PrevW-\PrevH-x}}%
          \edef\TempH{\getvalue{clips-\PrevW-\PrevH-y}}%
          \edef\CurrW{\getvalue{clips-\CurrentW-\CurrentH-x}}%
          \edef\CurrH{\getvalue{clips-\CurrentW-\CurrentH-y}}%
          \expanded{\setevalue{clips-\PrevW-\PrevH-x}{\CurrW}}%
          \expanded{\setevalue{clips-\PrevW-\PrevH-y}{\CurrH}}%
          \expanded{\setevalue{clips-\CurrentW-\CurrentH-x}{\TempW}}%
          \expanded{\setevalue{clips-\CurrentW-\CurrentH-y}{\TempH}}%
          \let\PrevW\CurrentW
          \let\PrevH\CurrentH
          \increment\CurrentI
        \fi
        \ifnum\CurrentN=\CurrentI\relax
          \exitloop
        \fi}%
   \fi
   \vbox
     {\forgetall
      \offinterlineskip
      \dorecurse{\MaxH}
        {\let\ClipH\recurselevel
         \hbox
           {\dorecurse{\MaxW}
              {\let\ClipW\recurselevel
               \ifnum\ClipH=\CurrentH\relax
                 \ifnum\ClipW=\CurrentW\relax
                   \def\HideClip{yes}%
                   \def\ClipColor{blue}%
                 \else
                   \def\HideClip{no}%
                   \def\ClipColor{gray}%
                 \fi
               \else
                 \def\HideClip{no}%
                 \def\ClipColor{gray}%
               \fi
               \framed
                 [width=fit,height=fit,offset=none,strut=no,
                  rulethickness=2pt,framecolor=blue,
                  background=color,backgroundcolor=\ClipColor,
                  empty=\HideClip]
                 {\clip
                    [x=\getvalue{clips-\ClipW-\ClipH-x},
                     y=\getvalue{clips-\ClipW-\ClipH-y}]
                    {\externalfigure[#2]}}}}}}%
   \egroup}

\def\NameBanner%
  {\color[gray]{\hbox to \hsize
     {\SmaFont \setstrut \strut
      \hss
      \ifnum\NameCounter=\NameLength\relax
        \gdef\NameCounter{1}%
      \else
        \doglobal\increment\NameCounter\relax
      \fi
      \getfromcommacommand[\NameList][\NameCounter]%
      \commalistelement\hss
      \dorecurse{2}
        {\ifnum\NameCounter=\NameLength\relax
           \def\NameCounter{1}%
         \else
           \increment\NameCounter\relax
         \fi
         \getfromcommacommand[\NameList][\NameCounter]%
         \commalistelement\hss}}}}
